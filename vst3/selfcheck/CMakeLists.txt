cmake_minimum_required(VERSION 3.20)
project(IrisSelfCheck LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(selfcheck ${CMAKE_CURRENT_SOURCE_DIR}/main.cpp)
target_include_directories(selfcheck PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../iris/source)
if(MSVC)
  target_compile_options(selfcheck PRIVATE /utf-8)
endif()
target_link_libraries(selfcheck PRIVATE gdiplus)
